% Allow syntax-highlighted notebook code to wrap within the page margins.
\usepackage{fvextra}
\fvset{
  breaklines=true,
  breakanywhere=true,
  breakautoindent=true
}
\RecustomVerbatimEnvironment{Highlighting}{Verbatim}{
  commandchars=\\\{\},
  breaklines=true,
  breakanywhere=true,
  breakautoindent=true
}
\renewenvironment{verbatim}{%
  \VerbatimEnvironment%
  \begin{Verbatim}[breaklines=true,breakanywhere=true,breakautoindent=true]%
}{%
  \end{Verbatim}%
}
